import React, { useState, useEffect, useCallback } from 'react';
import {
  View,
  Text,
  StyleSheet,
  ScrollView,
  TextInput,
  FlatList,
  Image,
  TouchableOpacity,
  ActivityIndicator,
  SafeAreaView,
  RefreshControl,
  Modal,
  Linking,
} from 'react-native';
import { Stack, useRouter } from 'expo-router';
import { Ionicons } from '@expo/vector-icons';
import { getAllLivros, addFavoriteBook } from '../../api/api';

const BookDetailModal = ({ book, visible, onClose }) => {
  if (!book) return null;
  const handleAddToFavorites = async () => {
    try {
      await addFavoriteBook(book._id);
      alert(`${book.titulo} foi adicionado aos favoritos!`);
      onClose();
    } catch (error) {
      console.error('Erro ao adicionar aos favoritos:', error);
      alert('Não foi possível adicionar aos favoritos.');
    }
  };
  const handleDownload = () => {
    if (book.download_link) {
      Linking.openURL(book.download_link).catch((err) =>
        console.error('Não foi possível abrir o link', err)
      );
    } else {
      alert('Link de download não disponível.');
    }
  };

  return (
    <Modal
      animationType='slide'
      transparent={true}
      visible={visible}
      onRequestClose={onClose}
    >
      <View style={modalStyles.backdrop}>
        <View style={modalStyles.contentContainer}>
          {/* CABEÇALHO DO MODAL */}
          <View style={modalStyles.header}>
            <Text style={modalStyles.title} numberOfLines={2}>
              {book.titulo}
            </Text>
            <TouchableOpacity onPress={onClose} style={modalStyles.closeButton}>
              <Ionicons name='close' size={28} color='#666' />
            </TouchableOpacity>
          </View>

          {/* ÁREA DE CONTEÚDO ROLÁVEL */}

          <View style={modalStyles.coverContainer}>
            <Image
              source={{ uri: book.imagem_uri }}
              style={modalStyles.coverImage}
            />
          </View>
          <Text style={modalStyles.author}>{book.autores}</Text>

          <View style={modalStyles.detailsGrid}>
            <Text style={modalStyles.detailItem}>
              **Páginas:** {book.pag || 'N/A'}
            </Text>
            <Text style={modalStyles.detailItem}>
              **Publicado:**{' '}
              {book.publiData ? new Date(book.publiData).getFullYear() : 'N/A'}
            </Text>
          </View>
          <ScrollView
            contentContainerStyle={modalStyles.scrollContent}
            showsVerticalScrollIndicator={false}
          >
            <Text style={modalStyles.description}>{book.descricao}</Text>
          </ScrollView>

          {/* BOTÕES FIXOS NA PARTE INFERIOR */}
          <View style={modalStyles.buttonContainer}>
            <TouchableOpacity
              style={[modalStyles.button, modalStyles.favoriteButton]}
              onPress={handleAddToFavorites}
            >
              <Ionicons name='heart' size={20} color='#fff' />
              <Text style={modalStyles.buttonText}>Favoritar</Text>
            </TouchableOpacity>
            <TouchableOpacity
              style={[modalStyles.button, modalStyles.downloadButton]}
              onPress={handleDownload}
            >
              <Ionicons name='download' size={20} color='#fff' />
              <Text style={modalStyles.buttonText}>Baixar</Text>
            </TouchableOpacity>
          </View>
        </View>
      </View>
    </Modal>
  );
};

const BookCard = ({ book, onCardPress }) => {
  return (
    <TouchableOpacity style={styles.bookCard} onPress={() => onCardPress(book)}>
      <Image
        source={{
          uri:
            book.imagem_uri ||
            'https://via.placeholder.com/140x180.png?text=Capa',
        }}
        style={styles.bookCover}
      />
      <Text style={styles.bookTitle} numberOfLines={1}>
        {book.titulo}
      </Text>
      <Text style={styles.bookAuthor} numberOfLines={1}>
        {book.autores}
      </Text>
    </TouchableOpacity>
  );
};

const HomeScreen = () => {
  const [todosLivros, setTodosLivros] = useState([]);
  const [livrosExibidos, setLivrosExibidos] = useState([]);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState(null);
  const [refreshing, setRefreshing] = useState(false);
  const [searchText, setSearchText] = useState('');
  const [activeFilter, setActiveFilter] = useState('Todos');
  const [modalVisible, setModalVisible] = useState(false);
  const [selectedBook, setSelectedBook] = useState(null);
  const router = useRouter();
  const filtros = ['Todos', 'Ficção Científica', 'Educacional', 'Fantasia'];

  const fetchLivros = useCallback(async () => {
    try {
      setError(null);
      const response = await getAllLivros();
      setTodosLivros(response.data);
      setLivrosExibidos(response.data);
    } catch (err) {
      setError('Não foi possível carregar os livros.');
      console.error('Erro ao buscar livros:', err);
    }
  }, []);
  useEffect(() => {
    const loadInitialData = async () => {
      setLoading(true);
      await fetchLivros();
      setLoading(false);
    };
    loadInitialData();
  }, [fetchLivros]);
  const onRefresh = useCallback(async () => {
    setRefreshing(true);
    await fetchLivros();
    setRefreshing(false);
  }, [fetchLivros]);
  useEffect(() => {
    let l = todosLivros;
    if (searchText) {
      l = l.filter(
        (i) =>
          i.titulo.toLowerCase().includes(searchText.toLowerCase()) ||
          i.autores.toLowerCase().includes(searchText.toLowerCase())
      );
    }
    if (activeFilter !== 'Todos') {
      l = l.filter((i) =>
        i.categoria.some(
          (c) => c.nome.toLowerCase() === activeFilter.toLowerCase()
        )
      );
    }
    setLivrosExibidos(l);
  }, [searchText, activeFilter, todosLivros]);
  const handleFilterPress = (f) => {
    setActiveFilter(f);
  };
  const handleCardPress = (b) => {
    setSelectedBook(b);
    setModalVisible(true);
  };
  const handleCloseModal = () => {
    setModalVisible(false);
    setSelectedBook(null);
  };

  if (loading && !refreshing) {
    return (
      <SafeAreaView style={[styles.container, styles.centered]}>
        <ActivityIndicator size='large' color='#6247AA' />
      </SafeAreaView>
    );
  }
  if (error) {
    return (
      <SafeAreaView style={[styles.container, styles.centered]}>
        <Text style={styles.errorText}>{error}</Text>
      </SafeAreaView>
    );
  }

  return (
    <SafeAreaView style={styles.container}>
      <Stack.Screen
        options={{
          headerTitle: 'Início',
          headerRight: () => (
            <View style={{ flexDirection: 'row' }}>
              {/* Ícone de Favoritos */}
              <TouchableOpacity
                onPress={() => router.push('/favoritos')}
                style={{ marginRight: 20 }}
              >
                <Ionicons name='heart-outline' size={30} color='#fff' />
              </TouchableOpacity>
              {/* Ícone de Perfil */}
              <TouchableOpacity onPress={() => router.push('/perfil')}>
                <Ionicons
                  name='person-circle-outline'
                  size={32}
                  color='#fff'
                  style={{ marginRight: 15 }}
                />
              </TouchableOpacity>
            </View>
          ),
        }}
      />
      <ScrollView
        showsVerticalScrollIndicator={false}
        refreshControl={
          <RefreshControl
            refreshing={refreshing}
            onRefresh={onRefresh}
            tintColor='#6247AA'
          />
        }
      >
        <View style={styles.searchContainer}>
          <Ionicons
            name='search'
            size={20}
            color='#888'
            style={styles.searchIcon}
          />
          <TextInput
            placeholder='Pesquisar por título ou autor...'
            style={styles.searchInput}
            value={searchText}
            onChangeText={setSearchText}
          />
          {searchText.length > 0 && (
            <TouchableOpacity onPress={() => setSearchText('')}>
              <Ionicons name='close-circle' size={20} color='#888' />
            </TouchableOpacity>
          )}
        </View>
        <View style={styles.filterContainer}>
          <ScrollView
            horizontal
            showsHorizontalScrollIndicator={false}
            contentContainerStyle={{ paddingHorizontal: 15 }}
          >
            {filtros.map((f) => (
              <TouchableOpacity
                key={f}
                style={[
                  styles.filterButton,
                  activeFilter === f && styles.filterButtonActive,
                ]}
                onPress={() => handleFilterPress(f)}
              >
                <Text
                  style={[
                    styles.filterButtonText,
                    activeFilter === f && styles.filterButtonTextActive,
                  ]}
                >
                  {f}
                </Text>
              </TouchableOpacity>
            ))}
          </ScrollView>
        </View>
        <View style={styles.section}>
          <Text style={styles.sectionTitle}>Livros Disponíveis</Text>
          <FlatList
            data={livrosExibidos}
            renderItem={({ item }) => (
              <BookCard book={item} onCardPress={handleCardPress} />
            )}
            keyExtractor={(item) => item._id.toString()}
            horizontal
            showsHorizontalScrollIndicator={false}
            contentContainerStyle={styles.listContainer}
            ListEmptyComponent={
              <Text style={styles.emptyListText}>Nenhum livro encontrado.</Text>
            }
          />
        </View>
      </ScrollView>
      <BookDetailModal
        book={selectedBook}
        visible={modalVisible}
        onClose={handleCloseModal}
      />
    </SafeAreaView>
  );
};
